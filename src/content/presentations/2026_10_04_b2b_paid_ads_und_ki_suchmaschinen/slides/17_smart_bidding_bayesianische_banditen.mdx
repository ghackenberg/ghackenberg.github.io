{`# Multi-Armed Bandit Bidding (Thompson Sampling)
# Probabilistische Gebotsfindung im Smart Bidding

import numpy as np

def calculate_dynamic_bid(
    alpha_conv: int, 
    beta_nonconv: int, 
    target_cpa: float
) -> float:
    """Zieht Konversions-Wahrscheinlichkeit aus Beta-Posterior."""
    p_conv = np.random.beta(alpha_conv, beta_nonconv)
    
    # Dynamisches Echtzeit-Gebot:
    # bid = target_cpa * p(conversion|context)
    optimal_bid = target_cpa * p_conv
    return round(optimal_bid, 2)`}
