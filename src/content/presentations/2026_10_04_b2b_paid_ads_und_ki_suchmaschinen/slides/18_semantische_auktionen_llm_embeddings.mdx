{`# Semantic Vector Matching in Real-Time Auctions
# Projektion von Query und Ad in dichte Embedding-Räume

import numpy as np

def cosine_similarity(
    query_emb: np.ndarray, 
    ad_emb: np.ndarray
) -> float:
    """Berechnet semantische Relevanz im Vektorraum."""
    dot = np.dot(query_emb, ad_emb)
    norm = np.linalg.norm(query_emb) * np.linalg.norm(ad_emb)
    return float(dot / (norm + 1e-9))

# Auktions-Trigger bei Kosinus-Ähnlichkeit >= Schwellenwert:
