{`# GSP Pricing Rule (Edelman et al. 2007 / Varian 2007)
# Bieter auf Position i zahlt das Mindestgebot zur Wahrung von Rang i

def calculate_gsp_cpc(
    ad_rank_next: float, 
    quality_score_current: float
) -> float:
    """Berechnet tatsächlichen Klickpreis (P_i) im GSP-Gleichgewicht."""
    min_bid = (ad_rank_next / quality_score_current) + 0.01
    return round(min_bid, 2)

# Symmetric Nash Equilibrium (SNE / lokal neidfrei):
# Nutzen(Position i) >= Nutzen(Position j) für alle j`}
